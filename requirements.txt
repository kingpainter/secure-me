# Pinned: this package bundles a specific Home Assistant core version for tests.

# predates 2024.11 -- before AlarmControlPanelState and ARM_VACATION existed --
# which breaks any test that touches alarm_control_panel.alarm_state.

# requires Python 3.13, which our CI matrix doesn't use). It bundles HA Core
# 2025.1, comfortably past the 2024.11 cutoff. If the CI Python matrix is ever
# bumped to 3.13+, this pin can be raised to track a newer HA core too.
pytest-homeassistant-custom-component==0.13.205
